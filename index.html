// ═══════════════════════════════════════════════════════════════ // ROTINA — Apps Script com suporte JSONP // Funciona tanto de file:// (local) quanto de https:// (GitHub Pages) // ═══════════════════════════════════════════════════════════════ const SHEET_TAREFAS = 'tarefas'; const SHEET_DIARIO = 'diario'; const SHEET_CONFIG = 'config'; const SHEET_BORDO = 'bordo'; // ─────────────────────────────────────────────────────────────── // ENTRY POINTS // ─────────────────────────────────────────────────────────────── function doGet(e) { const p = (e && e.parameter) ? e.parameter : {}; const action = p.action || 'load'; const cb = p.callback || ''; // JSONP callback name let result; if (action === 'ping') { result = { ok: true, ts: new Date().toISOString() }; } else if (action === 'load') { result = loadAll(); } else if (action === 'debug') { const ss = SpreadsheetApp.getActiveSpreadsheet(); result = { ok: true, sheets: ss.getSheets().map(s => s.getName()), tarefas: loadTarefas(ss), diario_keys: Object.keys(loadDiario(ss)), }; // ── Ações de escrita via GET/JSONP (payload como parâmetro) ── } else if (action === 'save_tarefas') { const payload = safeParseJSON(p.payload || '{}'); result = saveTarefas(payload.tarefas || []); } else if (action === 'save_diario') { const payload = safeParseJSON(p.payload || '{}'); result = saveDiario(payload); } else if (action === 'save_config') { const payload = safeParseJSON(p.payload || '{}'); result = saveConfig(payload); } else if (action === 'save_bordo') { const payload = safeParseJSON(p.payload || '{}'); result = saveBordo(payload); } else { result = { ok: false, error: 'Ação desconhecida: ' + action }; } return jsonpOrJson(result, cb); } // doPost mantido para compatibilidade com chamadas https:// function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonpOrJson({ ok: false, error: 'JSON inválido: ' + err.message }, ''); } const action = body.action || ''; const payload = body.payload || {}; let result; if (action === 'save_tarefas') result = saveTarefas(payload.tarefas || []); else if (action === 'save_diario') result = saveDiario(payload); else if (action === 'save_config') result = saveConfig(payload); else if (action === 'save_bordo') result = saveBordo(payload); else result = { ok: false, error: 'Ação POST desconhecida: ' + action }; return jsonpOrJson(result, ''); } // ─────────────────────────────────────────────────────────────── // LOAD ALL // ─────────────────────────────────────────────────────────────── function loadAll() { const ss = SpreadsheetApp.getActiveSpreadsheet(); return { ok: true, tarefas: loadTarefas(ss), diario: loadDiario(ss), config: loadConfig(ss), bordo: loadBordo(ss), }; } // ─────────────────────────────────────────────────────────────── // TAREFAS — colunas: id | nome | hora | freq | dia | data | cor | obs | tipo // ─────────────────────────────────────────────────────────────── function saveTarefas(tarefas) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = getOrCreateSheet(ss, SHEET_TAREFAS); const header = ['id', 'nome', 'hora', 'freq', 'dia', 'data', 'cor', 'obs', 'tipo']; sheet.clearContents(); sheet.getRange(1, 1, 1, header.length).setValues([header]); if (!tarefas.length) return { ok: true }; const rows = tarefas.map(t => [ String(t.id || ''), String(t.nome || ''), String(t.hora || ''), String(t.freq || 'daily'), String(t.dia || ''), String(t.data || ''), String(t.cor || 'blue'), String(t.obs || ''), String(t.tipo || 'tarefa'), ]); sheet.getRange(2, 1, rows.length, header.length).setValues(rows); return { ok: true }; } function loadTarefas(ss) { const sheet = ss.getSheetByName(SHEET_TAREFAS); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length < 2) return []; const header = data[0].map(h => String(h).trim().toLowerCase()); const get = (row, col) => { const i = header.indexOf(col); return i >= 0 ? String(row[i] || '') : ''; }; return data.slice(1).map(row => ({ id: get(row, 'id'), nome: get(row, 'nome'), hora: get(row, 'hora'), freq: get(row, 'freq') || 'daily', dia: get(row, 'dia'), data: normData(get(row, 'data')), cor: get(row, 'cor') || 'blue', obs: get(row, 'obs'), tipo: get(row, 'tipo') || 'tarefa', })).filter(t => t.id && t.nome); } // ─────────────────────────────────────────────────────────────── // DIÁRIO — colunas: data | checks_json | note | notas_json // ─────────────────────────────────────────────────────────────── function saveDiario(payload) { const { data, checks, note, notas } = payload; if (!data) return { ok: false, error: 'data ausente' }; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = getOrCreateSheet(ss, SHEET_DIARIO); ensureHeader(sheet, ['data', 'checks_json', 'note', 'notas_json']); const allData = sheet.getDataRange().getValues(); const header = allData[0].map(h => String(h).trim().toLowerCase()); const ci = col => header.indexOf(col); let targetRow = -1; for (let i = 1; i < allData.length; i++) { if (String(allData[i][ci('data')]) === String(data)) { targetRow = i + 1; break; } } const checksStr = JSON.stringify(checks || {}); const notasStr = JSON.stringify(notas || {}); const noteStr = String(note || ''); if (targetRow > 0) { sheet.getRange(targetRow, ci('checks_json') + 1).setValue(checksStr); sheet.getRange(targetRow, ci('note') + 1).setValue(noteStr); sheet.getRange(targetRow, ci('notas_json') + 1).setValue(notasStr); } else { const newRow = new Array(header.length).fill(''); newRow[ci('data')] = String(data); newRow[ci('checks_json')] = checksStr; newRow[ci('note')] = noteStr; newRow[ci('notas_json')] = notasStr; sheet.appendRow(newRow); } return { ok: true }; } function loadDiario(ss) { const sheet = ss.getSheetByName(SHEET_DIARIO); if (!sheet) return {}; const data = sheet.getDataRange().getValues(); if (data.length < 2) return {}; const header = data[0].map(h => String(h).trim().toLowerCase()); const ci = col => header.indexOf(col); const result = {}; data.slice(1).forEach(row => { const iso = String(row[ci('data')] || '').trim(); if (!iso || iso.length < 10) return; let checks = {}, notas = {}; try { checks = JSON.parse(row[ci('checks_json')] || '{}'); } catch(e) {} try { notas = JSON.parse(row[ci('notas_json')] || '{}'); } catch(e) {} result[iso] = { checks, note: String(row[ci('note')] || ''), notas }; }); return result; } // ─────────────────────────────────────────────────────────────── // CONFIG — colunas: chave | valor // ─────────────────────────────────────────────────────────────── function saveConfig(payload) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = getOrCreateSheet(ss, SHEET_CONFIG); ensureHeader(sheet, ['chave', 'valor']); Object.entries(payload).forEach(([chave, valor]) => { const rows = sheet.getDataRange().getValues(); let found = false; for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(chave)) { sheet.getRange(i+1,2).setValue(String(valor)); found=true; break; } } if (!found) sheet.appendRow([String(chave), String(valor)]); }); return { ok: true }; } function loadConfig(ss) { const sheet = ss.getSheetByName(SHEET_CONFIG); if (!sheet) return {}; const data = sheet.getDataRange().getValues(); const result = {}; data.slice(1).forEach(row => { if (row[0]) result[String(row[0])] = String(row[1]||''); }); return result; } // ─────────────────────────────────────────────────────────────── // BORDO — colunas: data | bordo_json // ─────────────────────────────────────────────────────────────── function saveBordo(payload) { const { data, entry } = payload; if (!data) return { ok: false, error: 'data ausente' }; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = getOrCreateSheet(ss, SHEET_BORDO); ensureHeader(sheet, ['data', 'bordo_json']); const allData = sheet.getDataRange().getValues(); const entryStr = JSON.stringify(entry || {}); let targetRow = -1; for (let i = 1; i < allData.length; i++) { if (String(allData[i][0]) === String(data)) { targetRow = i+1; break; } } if (targetRow > 0) sheet.getRange(targetRow, 2).setValue(entryStr); else sheet.appendRow([String(data), entryStr]); return { ok: true }; } function loadBordo(ss) { const sheet = ss.getSheetByName(SHEET_BORDO); if (!sheet) return {}; const data = sheet.getDataRange().getValues(); if (data.length < 2) return {}; const result = {}; data.slice(1).forEach(row => { const iso = String(row[0]||'').trim(); if (!iso) return; try { result[iso] = JSON.parse(row[1]||'{}'); } catch(e) { result[iso]={}; } }); return result; } // ─────────────────────────────────────────────────────────────── // UTILITÁRIOS // ─────────────────────────────────────────────────────────────── // Responde como JSONP se tiver callback, senão JSON puro function jsonpOrJson(obj, cb) { const body = JSON.stringify(obj); if (cb && cb.trim()) { return ContentService .createTextOutput(cb + '(' + body + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(body) .setMimeType(ContentService.MimeType.JSON); } function getOrCreateSheet(ss, name) { return ss.getSheetByName(name) || ss.insertSheet(name); } function ensureHeader(sheet, cols) { const first = sheet.getRange(1, 1, 1, cols.length).getValues()[0]; if (!first[0] || String(first[0]).trim() === '') { sheet.getRange(1, 1, 1, cols.length).setValues([cols]); } } function safeParseJSON(str) { try { return JSON.parse(str); } catch(e) { return {}; } } function normData(v) { if (!v) return ''; const s = String(v).trim(); if (!s || s.startsWith('1899') || s.startsWith('1900')) return ''; if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s; const m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/); if (m) return m[1]+'-'+m[2].padStart(2,'0')+'-'+m[3].padStart(2,'0'); return ''; }
